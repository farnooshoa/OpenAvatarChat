🔧 DEPLOYMENT STEPS (Method 2: Manual):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 1. Install HuggingFace CLI
pip install huggingface_hub

# 2. Login to HuggingFace
huggingface-cli login

# 3. Update submodules
git submodule update --init --recursive --depth 1

# 4. Copy deployment files
cp Dockerfile.huggingface Dockerfile
cp README_SPACE.md README.md
cp chat_with_groq_enhanced.yaml config/

# 5. Add HuggingFace remote (replace YOUR_USERNAME and YOUR_SPACE)
git remote add hf https://huggingface.co/spaces/YOUR_USERNAME/YOUR_SPACE

# 6. Commit and push
git add .
git commit -m "Deploy enhanced OpenAvatarChat to HuggingFace Spaces"
git push hf HEAD:main --force

🐛 TROUBLESHOOTING:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Problem: "Module not found"
Fix:     git submodule update --init --recursive --depth 1

Problem: "API key not working"
Fix:     Check Space Settings → Repository secrets
         Restart Space after adding secret

Problem: "Build timeout"
Fix:     Consider upgrading to paid tier for faster builds

Problem: "Port already in use"
Fix:     Ensure Dockerfile exposes 7860
         Check GRADIO_SERVER_PORT=7860 in environment

Problem: "Slow performance"
Fix:     Upgrade to GPU hardware in Space settings
